#include<stdio.h>
#include<stdlib.h>
struct Node {
    int data;
    struct Node* next;
};
struct Node *top=NULL;
void push(int value) {
    struct Node*newnode=(struct Node*)malloc(sizeof(struct Node));
    newnode->data=value;
    newnode->next=top;
    top=newnode;
    printf("%d pushed into stack.", value);
}
void pop() {
    struct Node*temp;
    if(top == NULL) {
        printf("stack under flow");
        return;
    }
    temp=top;
    printf("%d popped from stack.", top->data);
    top=top->next;
    free(temp);
}
void peek() { 
    if(top == NULL){
        printf("Stack is empty.\n");
    }
    else {
        printf("Top element=%d\n",top->data);
    }
}
void display() { 
    if(top == NULL) {
        printf("stack is empty.\n");
        return;
    }
    struct Node* temp=top;
    printf("Stack:");
    while (temp!=NULL) { 
        printf(" %d", temp->data);
        temp=temp->next;
    }
    printf("\n");
}
int main() { 
    push(10);
    push(20);
    push(30);
    display();
    peek();
    pop();
    return 0;
}
