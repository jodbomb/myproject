#include <stdio.h>
#include <stdlib.h>
struct Node { int data; struct Node* next; };
struct Node* tail = NULL;
void insertAtBeginning() { 
    int value; printf("Enter value to insert at beginning: "); scanf("%d", &value);
    struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));
    if (newNode == NULL) { printf("Memory allocation failed!\n"); return; }
    newNode->data = value;
    if (tail == NULL) {
        tail = newNode;
        tail->next = tail;
    } else {
        newNode->next = tail->next;
        tail->next = newNode;
    }
    printf("Node with value %d inserted.\n", value);
}
void insertAtEnd() { 
    int value; printf("Enter value to insert at end: "); scanf("%d", &value);
    struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));
    if (newNode == NULL) { printf("Memory allocation failed!\n"); return; }
    newNode->data = value;
    if (tail == NULL) {
        tail = newNode;
        tail->next = tail;
    } else {
        newNode->next = tail->next;
        tail->next = newNode;
        tail = newNode;
    }
    printf("Node with value %d inserted.\n", value);
}
void deleteNode() { 
    if (tail == NULL) { printf("List is empty! Nothing to delete.\n"); return; }
    int value;
    printf("Enter value to delete: "); scanf("%d", &value);
    struct Node* temp = tail->next;
    struct Node* prev = tail;
    if (temp == tail && temp->data == value) {
        free(temp);
        tail = NULL;
        printf("Node with value %d deleted.\n", value);
        return;
    }
    do {
        if (temp->data == value) {
            prev->next = temp->next;
            if (temp == tail) {
                tail = prev;
            }
            free(temp);
            printf("Node with value %d deleted.\n", value);
            return;
        }
        prev = temp;
        temp = temp->next;
    } while (temp != tail->next);
    printf("Value %d not found in the list.\n", value);
}
void display() { 
    if (tail == NULL) { printf("Circular Linked List is Empty.\n"); return; }
    struct Node* temp = tail->next;
    printf("Circular Linked List elements: ");
    do {
        printf("%d -> ", temp->data);
        temp = temp->next;
    } while (temp != tail->next);
    printf("(HEAD)\n");
}
int main() { 
    int choice;
    do {
        printf("\n--- SINGLY CIRCULAR LINKED LIST MENU ---\n");
        printf("1. Insert at Beginning\n");
        printf("2. Insert at End\n");
        printf("3. Delete Node by Value\n");
        printf("4. Display List\n");
        printf("5. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);
        switch (choice) {
            case 1: insertAtBeginning(); break;
            case 2: insertAtEnd(); break;
            case 3: deleteNode(); break;
            case 4: display(); break;
            case 5: printf("Exiting program...\n"); break;
            default: printf("Invalid choice! Try again.\n");
        }
    } while (choice != 5);
    return 0;
}
